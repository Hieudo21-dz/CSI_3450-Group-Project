IF DB_ID('FreeBee') IS NULL CREATE DATABASE FreeBee;

USE FreeBee;

/* Drop child tables first so the script can be re-run */
DROP TABLE IF EXISTS ORDER_ITEM, ORDERS, MENU_ALLERGEN, CUSTOMER_ALLERGY,
                     MENU_ITEM, CUSTOMER, ALLERGEN;
GO

CREATE TABLE ALLERGEN (
    AllergenID    INT IDENTITY(1,1) CONSTRAINT PK_ALLERGEN PRIMARY KEY,
    AllergenName  VARCHAR(50)   NOT NULL CONSTRAINT UQ_ALLERGEN_NAME UNIQUE,
    SafetyNote    VARCHAR(255)  NULL
);

CREATE TABLE CUSTOMER (
    CustomerID  INT IDENTITY(1,1) CONSTRAINT PK_CUSTOMER PRIMARY KEY,
    FirstName   VARCHAR(50)   NOT NULL,
    LastName    VARCHAR(50)   NOT NULL,
    Phone       VARCHAR(20)   NULL,
    Email       VARCHAR(100)  NOT NULL CONSTRAINT UQ_CUSTOMER_EMAIL UNIQUE
);

/* Bridge table: CUSTOMER M:N ALLERGEN */
CREATE TABLE CUSTOMER_ALLERGY (
    CustomerID  INT          NOT NULL
        CONSTRAINT FK_CA_CUSTOMER REFERENCES CUSTOMER (CustomerID),
    AllergenID  INT          NOT NULL
        CONSTRAINT FK_CA_ALLERGEN REFERENCES ALLERGEN (AllergenID),
    Severity    VARCHAR(10)  NOT NULL
        CONSTRAINT CK_CA_SEVERITY CHECK (Severity IN ('Mild', 'Moderate', 'Severe')),
    CONSTRAINT PK_CUSTOMER_ALLERGY PRIMARY KEY (CustomerID, AllergenID)
);

CREATE TABLE MENU_ITEM (
    MenuItemID  INT IDENTITY(1,1) CONSTRAINT PK_MENU_ITEM PRIMARY KEY,
    ItemName    VARCHAR(100)  NOT NULL CONSTRAINT UQ_MENU_ITEM_NAME UNIQUE,
    Category    VARCHAR(10)   NOT NULL
        CONSTRAINT CK_MENU_CATEGORY CHECK (Category IN ('Appetizer', 'Entree', 'Side', 'Dessert', 'Beverage')),
    Price       DECIMAL(6,2)  NOT NULL CONSTRAINT CK_MENU_PRICE CHECK (Price > 0)
);

/* Bridge table: MENU_ITEM M:N ALLERGEN.
   An item with no rows here is free of all tracked allergens. */
CREATE TABLE MENU_ALLERGEN (
    MenuItemID  INT  NOT NULL
        CONSTRAINT FK_MA_MENU_ITEM REFERENCES MENU_ITEM (MenuItemID),
    AllergenID  INT  NOT NULL
        CONSTRAINT FK_MA_ALLERGEN  REFERENCES ALLERGEN (AllergenID),
    CONSTRAINT PK_MENU_ALLERGEN PRIMARY KEY (MenuItemID, AllergenID)
);

CREATE TABLE ORDERS (
    OrderID      INT IDENTITY(1,1) CONSTRAINT PK_ORDERS PRIMARY KEY,
    CustomerID   INT           NOT NULL
        CONSTRAINT FK_ORDERS_CUSTOMER REFERENCES CUSTOMER (CustomerID),
    OrderDate    DATETIME      NOT NULL CONSTRAINT DF_ORDERS_DATE   DEFAULT GETDATE(),
    TotalAmount  DECIMAL(8,2)  NOT NULL CONSTRAINT DF_ORDERS_TOTAL  DEFAULT 0
        CONSTRAINT CK_ORDERS_TOTAL CHECK (TotalAmount >= 0),
    OrderStatus  VARCHAR(10)   NOT NULL CONSTRAINT DF_ORDERS_STATUS DEFAULT 'Pending'
        CONSTRAINT CK_ORDERS_STATUS CHECK (OrderStatus IN ('Pending', 'Confirmed', 'Completed', 'Cancelled'))
);

/* Bridge table: ORDERS M:N MENU_ITEM */
CREATE TABLE ORDER_ITEM (
    OrderID     INT  NOT NULL
        CONSTRAINT FK_OI_ORDERS    REFERENCES ORDERS (OrderID),
    MenuItemID  INT  NOT NULL
        CONSTRAINT FK_OI_MENU_ITEM REFERENCES MENU_ITEM (MenuItemID),
    Quantity    INT  NOT NULL
        CONSTRAINT CK_OI_QUANTITY CHECK (Quantity BETWEEN 1 AND 50),
    CONSTRAINT PK_ORDER_ITEM PRIMARY KEY (OrderID, MenuItemID)
);
GO

/* ---------------------------------------------------------
   TRIGGER 1: Order Safety Verification
   Blocks an item from being added to an order when the item
   contains an allergen that the order's customer is allergic to.
   Raises error 50001 with a message the app can show as-is,
   and the insert/update is undone.
   --------------------------------------------------------- */
CREATE OR ALTER TRIGGER TRG_ORDER_ITEM_SAFETY
ON ORDER_ITEM
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @Msg NVARCHAR(400);

    /* Find a conflict (most severe first) and build the message */
    SELECT TOP (1)
           @Msg = CONCAT('Unsafe order blocked: ', M.ItemName, ' contains ', A.AllergenName,
                         ', and ', C.FirstName, ' ', C.LastName, ' has a ',
                         LOWER(CA.Severity), ' ', A.AllergenName, ' allergy.')
    FROM   inserted I
    JOIN   ORDERS O            ON O.OrderID     = I.OrderID
    JOIN   CUSTOMER C          ON C.CustomerID  = O.CustomerID
    JOIN   CUSTOMER_ALLERGY CA ON CA.CustomerID = O.CustomerID
    JOIN   MENU_ALLERGEN MA    ON MA.MenuItemID = I.MenuItemID
                              AND MA.AllergenID = CA.AllergenID
    JOIN   MENU_ITEM M         ON M.MenuItemID  = I.MenuItemID
    JOIN   ALLERGEN A          ON A.AllergenID  = CA.AllergenID
    ORDER BY CASE CA.Severity WHEN 'Severe' THEN 1 WHEN 'Moderate' THEN 2 ELSE 3 END;

    IF @Msg IS NOT NULL
    BEGIN
        THROW 50001, @Msg, 1;
    END
END
GO

/* ---------------------------------------------------------
   TRIGGER 2: keep ORDERS.TotalAmount correct
   Recalculates the order total whenever its items are added,
   changed or removed, so the app never has to send a total.
   --------------------------------------------------------- */
CREATE OR ALTER TRIGGER TRG_ORDER_ITEM_TOTAL
ON ORDER_ITEM
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE O
    SET    TotalAmount = ISNULL((SELECT SUM(OI.Quantity * M.Price)
                                 FROM   ORDER_ITEM OI
                                 JOIN   MENU_ITEM M ON M.MenuItemID = OI.MenuItemID
                                 WHERE  OI.OrderID = O.OrderID), 0)
    FROM   ORDERS O
    WHERE  O.OrderID IN (SELECT OrderID FROM inserted
                         UNION
                         SELECT OrderID FROM deleted);
END
GO

   report queries. Highlight one query and run it to get a
   screenshot of just that result.

USE FreeBee;
GO

/* ===================== VIEWS ===================== */

/* Every (customer, menu item) pair that is SAFE: the item contains
   none of the customer's allergens. The app filters by CustomerID. */
CREATE OR ALTER VIEW VW_SAFE_MENU AS
SELECT C.CustomerID, M.MenuItemID, M.ItemName, M.Category, M.Price
FROM   CUSTOMER C
CROSS JOIN MENU_ITEM M
WHERE  NOT EXISTS (SELECT 1
                   FROM   CUSTOMER_ALLERGY CA
                   JOIN   MENU_ALLERGEN MA ON MA.AllergenID = CA.AllergenID
                   WHERE  CA.CustomerID = C.CustomerID
                     AND  MA.MenuItemID = M.MenuItemID);
GO

/* One row per customer allergy, with names instead of IDs. */
CREATE OR ALTER VIEW VW_CUSTOMER_ALLERGIES AS
SELECT C.CustomerID, C.FirstName, C.LastName,
       A.AllergenID, A.AllergenName, CA.Severity, A.SafetyNote
FROM   CUSTOMER C
JOIN   CUSTOMER_ALLERGY CA ON CA.CustomerID = C.CustomerID
JOIN   ALLERGEN A          ON A.AllergenID  = CA.AllergenID;
GO

/* ===================== QUERIES THE APP USES ===================== */

-- Q1. What is this customer allergic to? (customer 1 as the example)
SELECT FirstName, LastName, AllergenName, Severity, SafetyNote
FROM   VW_CUSTOMER_ALLERGIES
WHERE  CustomerID = 1
ORDER BY CASE Severity WHEN 'Severe' THEN 1 WHEN 'Moderate' THEN 2 ELSE 3 END;

-- Q2. Which dishes are safe for this customer? (the filtered menu)
SELECT Category, ItemName, Price
FROM   VW_SAFE_MENU
WHERE  CustomerID = 1
ORDER BY Category, ItemName;

-- Q3. Which dishes must this customer avoid, and why?
SELECT M.ItemName, M.Category,
       STRING_AGG(A.AllergenName, ', ') WITHIN GROUP (ORDER BY A.AllergenName) AS ConflictingAllergens
FROM   CUSTOMER_ALLERGY CA
JOIN   MENU_ALLERGEN MA ON MA.AllergenID = CA.AllergenID
JOIN   MENU_ITEM M      ON M.MenuItemID  = MA.MenuItemID
JOIN   ALLERGEN A       ON A.AllergenID  = CA.AllergenID
WHERE  CA.CustomerID = 1
GROUP BY M.ItemName, M.Category
ORDER BY M.Category, M.ItemName;

/* ===================== REPORTS ===================== */

-- Q4. Which menu items are free of all tracked allergens? (outer join)
SELECT M.ItemName, M.Category, M.Price
FROM   MENU_ITEM M
LEFT JOIN MENU_ALLERGEN MA ON MA.MenuItemID = M.MenuItemID
WHERE  MA.MenuItemID IS NULL
ORDER BY M.Category, M.ItemName;

-- Q5. Which allergens are most common among our customers?
SELECT A.AllergenName,
       COUNT(CA.CustomerID)                                       AS Customers,
       SUM(CASE WHEN CA.Severity = 'Severe' THEN 1 ELSE 0 END)    AS SevereCases
FROM   ALLERGEN A
LEFT JOIN CUSTOMER_ALLERGY CA ON CA.AllergenID = A.AllergenID
GROUP BY A.AllergenName
ORDER BY Customers DESC, SevereCases DESC, A.AllergenName;

-- Q6. How much of the menu does each allergen affect?
SELECT A.AllergenName,
       COUNT(MA.MenuItemID) AS MenuItems,
       CAST(100.0 * COUNT(MA.MenuItemID) / (SELECT COUNT(*) FROM MENU_ITEM) AS DECIMAL(5,1)) AS PercentOfMenu
FROM   ALLERGEN A
LEFT JOIN MENU_ALLERGEN MA ON MA.AllergenID = A.AllergenID
GROUP BY A.AllergenName
ORDER BY MenuItems DESC, A.AllergenName;

-- Q7. Which menu items contain the most allergens, and which ones?
SELECT M.ItemName, M.Category,
       COUNT(*) AS AllergenCount,
       STRING_AGG(A.AllergenName, ', ') WITHIN GROUP (ORDER BY A.AllergenName) AS Allergens
FROM   MENU_ITEM M
JOIN   MENU_ALLERGEN MA ON MA.MenuItemID = M.MenuItemID
JOIN   ALLERGEN A       ON A.AllergenID  = MA.AllergenID
GROUP BY M.ItemName, M.Category
ORDER BY AllergenCount DESC, M.ItemName;

-- Q8. Which customers have a severe allergy? (staff alert list)
SELECT LastName, FirstName, AllergenName, SafetyNote
FROM   VW_CUSTOMER_ALLERGIES
WHERE  Severity = 'Severe'
ORDER BY LastName, FirstName, AllergenName;

-- Q9. How many safe dishes does each customer have to choose from?
--     (customers at the top have the fewest options)
SELECT C.CustomerID, C.FirstName, C.LastName,
       (SELECT COUNT(*) FROM CUSTOMER_ALLERGY CA WHERE CA.CustomerID = C.CustomerID) AS Allergies,
       COUNT(S.MenuItemID) AS SafeDishes
FROM   CUSTOMER C
LEFT JOIN VW_SAFE_MENU S ON S.CustomerID = C.CustomerID
GROUP BY C.CustomerID, C.FirstName, C.LastName
ORDER BY SafeDishes, C.LastName;

-- Q10. Sales by menu category (cancelled orders left out)
SELECT M.Category,
       SUM(OI.Quantity)           AS UnitsSold,
       SUM(OI.Quantity * M.Price) AS Revenue
FROM   ORDER_ITEM OI
JOIN   ORDERS O    ON O.OrderID    = OI.OrderID
JOIN   MENU_ITEM M ON M.MenuItemID = OI.MenuItemID
WHERE  O.OrderStatus <> 'Cancelled'
GROUP BY M.Category
ORDER BY Revenue DESC;

-- Q11. The five best-selling menu items
SELECT TOP 5 M.ItemName, M.Category,
       SUM(OI.Quantity)           AS UnitsSold,
       SUM(OI.Quantity * M.Price) AS Revenue
FROM   ORDER_ITEM OI
JOIN   ORDERS O    ON O.OrderID    = OI.OrderID
JOIN   MENU_ITEM M ON M.MenuItemID = OI.MenuItemID
WHERE  O.OrderStatus <> 'Cancelled'
GROUP BY M.ItemName, M.Category
ORDER BY UnitsSold DESC, Revenue DESC;

-- Q12. Customers who have spent more than the average customer (HAVING + subquery)
SELECT C.CustomerID, C.FirstName, C.LastName,
       COUNT(*)           AS Orders,
       SUM(O.TotalAmount) AS TotalSpent
FROM   CUSTOMER C
JOIN   ORDERS O ON O.CustomerID = C.CustomerID
WHERE  O.OrderStatus <> 'Cancelled'
GROUP BY C.CustomerID, C.FirstName, C.LastName
HAVING SUM(O.TotalAmount) > (SELECT AVG(T.Spent)
                             FROM (SELECT SUM(TotalAmount) AS Spent
                                   FROM   ORDERS
                                   WHERE  OrderStatus <> 'Cancelled'
                                   GROUP BY CustomerID) AS T)
ORDER BY TotalSpent DESC;

-- Q13. Orders and revenue per month
SELECT CONVERT(CHAR(7), OrderDate, 126) AS OrderMonth,
       COUNT(*)                         AS Orders,
       SUM(TotalAmount)                 AS Revenue,
       CAST(AVG(TotalAmount) AS DECIMAL(8,2)) AS AverageOrder
FROM   ORDERS
WHERE  OrderStatus <> 'Cancelled'
GROUP BY CONVERT(CHAR(7), OrderDate, 126)
ORDER BY OrderMonth;
